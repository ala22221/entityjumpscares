local Plr = Players.LocalPlayer

local admeen = {
    "alamgir_500newacc"
}

local function isAdmin(name)
    name = string.lower(tostring(name))

    for _, adminName in ipairs(admeen) do
        if string.lower(adminName) == name then
            return true
        end
    end

    return false
end

local IsAdmin = isAdmin(Plr.Name)

-- Captions

local DoorsCaptions

pcall(function()
    local source = game:HttpGet(
        "https://raw.githubusercontent.com/RegularVynixu/DOORS-Captions/main/init.luau"
    )

    local loader = loadstring(source)

    if loader then
        DoorsCaptions = loader()
    end
end)

local function caption(text, captionType, duration)
    if DoorsCaptions and type(DoorsCaptions.caption) == "function" then
        pcall(function()
            DoorsCaptions.caption(
                text,
                captionType or "info",
                duration or 5
            )
        end)
    end
end

-- Achievement

local AchievementGiver

pcall(function()
    local source = game:HttpGet(
        "https://raw.githubusercontent.com/ala22221/Backups-of-some-stuff/refs/heads/main/Fixed%20Achievements%20by%20DripCapybara.lua"
    )

    local loader = loadstring(source)

    if loader then
        local result = loader()

        if type(result) == "function" then
            AchievementGiver = result
        end
    end
end)

local function waitForAchievementUI()
    local playerGui = Plr:WaitForChild("PlayerGui", 15)

    if not playerGui then
        return false
    end

    local globalUI = playerGui:WaitForChild("GlobalUI", 15)

    if not globalUI then
        return false
    end

    local achievementsHolder = globalUI:WaitForChild(
        "AchievementsHolder",
        15
    )

    if not achievementsHolder then
        return false
    end

    local achievement = achievementsHolder:WaitForChild(
        "Achievement",
        15
    )

    return achievement ~= nil
end

local givenAchievements = {}

local function giveAchievement(id, data)
    if givenAchievements[id] then
        return
    end

    if not AchievementGiver then
        warn("Achievement giver failed to load.")
        return
    end

    if not waitForAchievementUI() then
        warn("Achievement UI was not found.")
        return
    end

    givenAchievements[id] = true

    task.spawn(function()
        pcall(function()
            AchievementGiver(data)
        end)
    end)
end

-- Spawner

local SpawnerURL =
    "https://raw.githubusercontent.com/ala22221/Backups-of-some-stuff/refs/heads/main/Fixed%20Spawner%20by%20DripCapybara.lua"

    'if CG:FindFirstChild%("JumpscareGui"%)%s*or%s*%(%s*Plr%.PlayerGui%.MainUI%.Death%.HelpfulDialogue%.Visible%s*and%s*not%s*Plr%.PlayerGui%.MainUI%.DeathPanelDead%.Visible%s*%) then',
    'if CG:FindFirstChild("JumpscareGui") then'
)

SpawnerSource = SpawnerSource:gsub(
    'Plr%.PlayerGui%.MainUI%.Death%.HelpfulDialogue%.Visible',
    'false'
)

local SpawnerLoader, SpawnerError = loadstring(SpawnerSource)

if not SpawnerLoader then
    error("Spawner compile error: " .. tostring(SpawnerError))
end

local Spawner = SpawnerLoader()

if type(Spawner) ~= "table" then

end

if type(Spawner.createEntity) ~= "function" then
    error("Spawner.createEntity is missing.")
end

if type(Spawner.runEntity) ~= "function" then
    error("Spawner.runEntity is missing.")

-- Admin Commands

local lastCommand = ""
local lastCommandTime = 0

local function handleCommand(message)

    if not IsAdmin then
        return
    end

    message = tostring(message):lower()

    message = message:match("^%s*(.-)%s*$")

    local now = os.clock()

    if message == lastCommand
        and now - lastCommandTime < 0.25 then
        return
    end

    lastCommand = message
    lastCommandTime = now

    if message == "/depth" then
        spawnDepth()

    elseif message == "/deergod" then
        spawnDeerGod()
    end
end

-- TextChatService

pcall(function()
    TextChatService.MessageReceived:Connect(
        function(message)

            if not message.TextSource then
                return
            end

            if message.TextSource.UserId
                ~= Plr.UserId then
                return
            end

            handleCommand(message.Text)
        end
    )
end)

-- Legacy Chat

pcall(function()
    Plr.Chatted:Connect(function(message)
        handleCommand(message)
    end)
end)
