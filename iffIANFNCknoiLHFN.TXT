    "alamgir_500newacc"
}

local function isAdmin(name)

    name = string.lower(tostring(name))

    for _, adminName in ipairs(admeen) do

        if string.lower(adminName) == name then
            return true
        end

    end

    return false
end

-- =========================================================
-- SeekMoving / Figure
-- =========================================================

local function makeFigureVisible(figure)

    -- Handle Figure itself

    if figure:IsA("BasePart") then

        figure.Transparency = 0

    elseif figure:IsA("Decal")
        or figure:IsA("Texture") then

        figure.Transparency = 0

    elseif figure:IsA("ParticleEmitter") then

        figure.Enabled = true

    end

    -- Handle everything inside Figure

    for _, obj in ipairs(figure:GetDescendants()) do

        if obj:IsA("BasePart") then

            obj.Transparency = 0

        elseif obj:IsA("Decal")
            or obj:IsA("Texture") then

            obj.Transparency = 0

        elseif obj:IsA("ParticleEmitter") then

            obj.Enabled = true

        end

    end
end

local function setupSeekMoving(model)

    if not model:IsA("Model") then
        return
    end

    -- Hide everything inside SeekMoving

    for _, obj in ipairs(model:GetDescendants()) do

        if obj:IsA("BasePart")
            or obj:IsA("Decal")
            or obj:IsA("Texture") then

            obj.Transparency = 1

        elseif obj:IsA("ParticleEmitter") then

            obj.Enabled = false

        end

    end

    -- Find Figure recursively

    local figure =
        model:FindFirstChild("Figure", true)

    if not figure then
        return
    end

    -- Make Figure visible

    makeFigureVisible(figure)

    -- Handle children added later

    figure.DescendantAdded:Connect(function(obj)

        if obj:IsA("BasePart")
            or obj:IsA("Decal")
            or obj:IsA("Texture") then

            obj.Transparency = 0

        elseif obj:IsA("ParticleEmitter") then

            obj.Enabled = true

        end

    end)
end

-- Handle SeekMoving models that already exist

for _, obj in ipairs(Workspace:GetDescendants()) do

    if obj:IsA("Model")
        and obj.Name == "SeekMoving" then

        task.spawn(function()

            setupSeekMoving(obj)

        end)

    end

end

-- Handle newly spawned SeekMoving models

Workspace.DescendantAdded:Connect(function(obj)

    if obj:IsA("Model")
        and obj.Name == "SeekMoving" then

        task.wait()

        setupSeekMoving(obj)

    end

end)

local function caption(text, captionType, duration)

    if DoorsCaptions
        and type(DoorsCaptions.caption) == "function" then

        pcall(function()

            DoorsCaptions.caption(
                text,
                captionType or "info",
                duration or 5
            )

        end)

    end
end

local AchievementGiver

pcall(function()

    local source = game:HttpGet(
        "https://raw.githubusercontent.com/ala22221/Backups-of-some-stuff/refs/heads/main/Fixed%20Achievements%20by%20DripCapybara.lua"
    )

    local loader = loadstring(source)

    if loader then

        local result = loader()

        if type(result) == "function" then
            AchievementGiver = result
        end

    end

end)

local function waitForAchievementUI()

    local playerGui =
        Plr:WaitForChild("PlayerGui", 15)

    if not playerGui then
        return false
    end

    local globalUI =
        playerGui:WaitForChild("GlobalUI", 15)

    if not globalUI then
        return false
    end

    local achievementsHolder =
        globalUI:WaitForChild(
            "AchievementsHolder",
            15
        )

    if not achievementsHolder then
        return false
    end

    local achievement =
        achievementsHolder:WaitForChild(
            "Achievement",
            15
        )

    return achievement ~= nil
end

local givenAchievements = {}

local function giveAchievement(id, data)

    if givenAchievements[id] then
        return
    end

    if not AchievementGiver then

        warn(
            "Achievement giver failed to load."
        )

        return
    end

    if not waitForAchievementUI() then

        warn(
            "Achievement UI was not found."
        )

        return
    end

    givenAchievements[id] = true

    task.spawn(function()

        pcall(function()

            AchievementGiver(data)

        end)

    end)
end

-- =========================================================
-- Admin Commands
-- =========================================================

local lastCommand = ""
local lastCommandTime = 0

local function handleCommand(message)

    if not IsAdmin then
        return
    end

    message =
        tostring(message):lower()

    message =
        message:match("^%s*(.-)%s*$")

    local now = os.clock()

    if message == lastCommand
        and now - lastCommandTime < 0.25 then

        return
    end

    lastCommand = message
    lastCommandTime = now

    if message == "/depth" then

        spawnDepth()

    elseif message == "/rebound" then

        spawnRebound()

    elseif message == "/deergod" then

        spawnDeerGod()

    end

end

-- =========================================================
-- TextChatService
-- =========================================================

pcall(function()

    TextChatService.MessageReceived:Connect(
        function(message)

            if not message.TextSource then
                return
            end

            if message.TextSource.UserId
                ~= Plr.UserId then

                return

            end

            handleCommand(
                message.Text
            )

        end
    )

end)

-- Legacy Chat

    )
