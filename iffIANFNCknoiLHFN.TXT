-- =========================================================
-- SeekMoving / Figure
-- Instant model changer during the cutscene
-- =========================================================

local SEEK_NAME = "SeekMovingNewClone"
local currentSeek = nil
local seekConnections = {}
local seekPartConnections = {}
local seekSetupToken = 0

local function disconnectSeekConnections()
    for _, connection in ipairs(seekConnections) do
        connection:Disconnect()
    end

    for _, connection in ipairs(seekPartConnections) do
        connection:Disconnect()
    end

    table.clear(seekConnections)
    table.clear(seekPartConnections)
end

local function applySeekChanges(seekClone)
    if not seekClone or not seekClone.Parent then
        return
    end

    local seekRig = seekClone:FindFirstChild("SeekRig")
    local figure = seekClone:FindFirstChild("Figure")

    -- Keep the replacement Figure visible.
    if figure and figure:IsA("BasePart") then
        figure.Transparency = 0

        local scream = figure:FindFirstChild("Scream")
        local flange = scream
            and scream:FindFirstChild("FlangeSoundEffect")

        if flange then
            flange.Enabled = false
        end

        local attachment = figure:FindFirstChild("Attachment")
        local emitter = attachment
            and attachment:FindFirstChild("ParticleEmitter")

        if emitter then
            emitter.Enabled = true
        end

        local camAttach = figure:FindFirstChild("CamAttach")
        local spotlight = camAttach
            and camAttach:FindFirstChild("SpotLight")

        if spotlight then
            spotlight.Enabled = true
        end
    end

    -- Hide the original SeekRig.
    if seekRig then
        for _, object in ipairs(seekRig:GetDescendants()) do
            if object:IsA("BasePart") then
                object.Transparency = 1
            elseif object:IsA("Decal")
                or object:IsA("Texture") then
                object.Transparency = 1
            end
        end
    end
end

local function watchSeekPart(object, seekClone)
    if not object:IsA("BasePart") then
        return
    end

    local connection = object:GetPropertyChangedSignal(
        "Transparency"
    ):Connect(function()
        if currentSeek ~= seekClone
            or not seekClone.Parent then
            return
        end

        -- Only correct transparency when the game changes it.
        task.defer(function()
            if currentSeek == seekClone
                and seekClone.Parent then
                applySeekChanges(seekClone)
            end
        end)
    end)

    table.insert(seekPartConnections, connection)
end

local function setupSeek(seekClone)
    if not seekClone or not seekClone.Parent then
        return
    end

    if currentSeek == seekClone then
        applySeekChanges(seekClone)
        return
    end

    disconnectSeekConnections()

    currentSeek = seekClone
    seekSetupToken += 1

    local setupToken = seekSetupToken

    -- Apply the changes immediately.
    applySeekChanges(seekClone)

    -- Watch existing parts for transparency resets.
    for _, object in ipairs(seekClone:GetDescendants()) do
        watchSeekPart(object, seekClone)
    end

    -- Catch parts that the cutscene adds later.
    table.insert(
        seekConnections,
        seekClone.DescendantAdded:Connect(function(object)
            if currentSeek ~= seekClone then
                return
            end

            watchSeekPart(object, seekClone)

            task.defer(function()
                if currentSeek == seekClone
                    and seekClone.Parent then
                    applySeekChanges(seekClone)
                end
            end)
        end)
    )

    -- Detect when the model is removed.
    table.insert(
        seekConnections,
        seekClone.AncestryChanged:Connect(function()
            if currentSeek ~= seekClone then
                return
            end

            if not seekClone.Parent then
                currentSeek = nil
                disconnectSeekConnections()
            end
        end)
    )

    -- A short startup retry catches parts that appear immediately
    -- after the model is parented to Workspace.
    task.spawn(function()
        for _ = 1, 10 do
            if currentSeek ~= seekClone
                or not seekClone.Parent
                or setupToken ~= seekSetupToken then
                return
            end

            applySeekChanges(seekClone)
            task.wait(0.1)
        end
    end)
end

-- Detect Seek spawning during the cutscene.
Workspace.ChildAdded:Connect(function(child)
    if child.Name == SEEK_NAME then
        setupSeek(child)
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child == currentSeek then
        currentSeek = nil
        seekSetupToken += 1
        disconnectSeekConnections()
    end
end)

-- Handle Seek if it already exists.
local existingSeek = Workspace:FindFirstChild(SEEK_NAME)

if existingSeek then
    setupSeek(existingSeek)
end

local function waitForAchievementUI()
    local playerGui = Plr:WaitForChild("PlayerGui", 15)

    if not playerGui then
        return false
    end

    local globalUI = playerGui:WaitForChild("GlobalUI", 15)

    if not globalUI then
        return false
    end

    local achievementsHolder = globalUI:WaitForChild(
        "AchievementsHolder",
        15
    )

    if not achievementsHolder then
        return false
    end

    local achievement = achievementsHolder:WaitForChild(
        "Achievement",
        15
    )

    return achievement ~= nil
end

local givenAchievements = {}

local function giveAchievement(id, data)
    if givenAchievements[id] then
        return
    end

    if not AchievementGiver then
        return
    end

    if not waitForAchievementUI() then
        return
    end

    givenAchievements[id] = true

    task.spawn(function()
        pcall(function()
            AchievementGiver(data)
        end)
    end)
end

local SeekMusicURL =
    "https://raw.githubusercontent.com/ala22221/sounds/main/106.ogg"

local SeekMusicFolder = "CustomMusic"
local SeekMusicFile = SeekMusicFolder .. "/106.ogg"
local SeekMusicVolume = 1

task.spawn(function()
    if not (
        writefile
        and isfile
        and getcustomasset
        and makefolder
        and isfolder
    ) then
        caption(
            "Executor doesn't support custom audio.",
            "info",
            5
        )
        return
    end

    pcall(function()
        if not isfolder(SeekMusicFolder) then
            makefolder(SeekMusicFolder)
        end

        if not isfile(SeekMusicFile) then
            local ok, data = pcall(function()
                return game:HttpGet(SeekMusicURL)
            end)

            if not ok or not data or #data == 0 then
                caption("Seek music download failed.", "info", 5)
                return
            end

            writefile(SeekMusicFile, data)
        end

        local assetId = getcustomasset(SeekMusicFile)

        local floorRep = ReplicatedStorage:WaitForChild(
            "FloorReplicated",
            30
        )

        if not floorRep then
            return
        end

        local hookedSounds = {}

        local function hookSound(sound)
            if not sound
                or not sound:IsA("Sound")
                or hookedSounds[sound] then
                return
            end

            hookedSounds[sound] = true

            local function applyReplacement()
                if not sound.Parent then
                    return
                end

                if sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end

                if sound.Volume ~= SeekMusicVolume then
                    sound.Volume = SeekMusicVolume
                end
            end

            applyReplacement()

            sound:GetPropertyChangedSignal("SoundId"):Connect(function()
                if sound.Parent and sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end
            end)

            sound:GetPropertyChangedSignal("Volume"):Connect(function()
                if sound.Parent
                    and sound.Volume ~= SeekMusicVolume then
                    sound.Volume = SeekMusicVolume
                end
            end)
        end

        local seekMusic = floorRep:FindFirstChild("SeekMusic")

        if seekMusic then
            hookSound(seekMusic)
        end

        floorRep.ChildAdded:Connect(function(child)
            if child.Name == "SeekMusic" then
                task.defer(function()
                    hookSound(child)
                end)
            end
        end)

        local function checkClone(obj)
            if obj:IsA("Sound") and obj.Name == "SeekMusic" then
                task.defer(function()
                    hookSound(obj)
                end)
            end
        end

        Workspace.DescendantAdded:Connect(checkClone)
        SoundService.DescendantAdded:Connect(checkClone)

        caption("music replacer loaded!", "info", 5)
    end)
end)

local BlueMusicVolume = 1

    end

    local ok, err = pcall(function()

        end

        if not isfile(BlueMusicFile) then
            local downloadOK, data = pcall(function()

            end)

            writefile(BlueMusicFile, data)
        end

        local function findBlueMusic()
            local playerGui = Plr:FindFirstChildOfClass("PlayerGui")

            if not playerGui then
                return nil
            end

            local mainUI = playerGui:FindFirstChild("MainUI")
            local initiator = mainUI
                and mainUI:FindFirstChild("Initiator")
            local mainGame = initiator
                and initiator:FindFirstChild("Main_Game")
            local health = mainGame
                and mainGame:FindFirstChild("Health")
            local music = health and health:FindFirstChild("Music")
            local blue = music and music:FindFirstChild("Blue")

            if blue and blue:IsA("Sound") then
                return blue
            end

            return nil
        end

        local hookedSounds = {}

        local function hookBlueMusic(sound)
            if not sound
                or not sound:IsA("Sound")
                or hookedSounds[sound] then
                return
            end

            hookedSounds[sound] = true

            local function applyReplacement()
                if not sound.Parent then
                    return
                end

                if sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end

                if sound.Volume ~= BlueMusicVolume then
                    sound.Volume = BlueMusicVolume
                end

                if sound.Looped ~= BlueMusicLooped then
                    sound.Looped = BlueMusicLooped
                end
            end

            applyReplacement()

            sound:GetPropertyChangedSignal("SoundId"):Connect(function()
                if sound.Parent and sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end
            end)

            sound:GetPropertyChangedSignal("Volume"):Connect(function()
                if sound.Parent
                    and sound.Volume ~= BlueMusicVolume then
                    sound.Volume = BlueMusicVolume
                end
            end)

            sound:GetPropertyChangedSignal("Looped"):Connect(function()
                if sound.Parent
                    and sound.Looped ~= BlueMusicLooped then
                    sound.Looped = BlueMusicLooped
                end
            end)

            pcall(function()
                sound:Play()
            end)
        end

        local playerGui = Plr:WaitForChild("PlayerGui", 30)

        if not playerGui then
            error("PlayerGui did not load.")
        end

        local blue = findBlueMusic()

        if not blue then
            local mainUI = playerGui:WaitForChild("MainUI", 30)
            local initiator = mainUI
                and mainUI:WaitForChild("Initiator", 30)
            local mainGame = initiator
                and initiator:WaitForChild("Main_Game", 30)
            local health = mainGame
                and mainGame:WaitForChild("Health", 30)
            local music = health
                and health:WaitForChild("Music", 30)

            blue = music and music:WaitForChild("Blue", 30)
        end

        if blue and blue:IsA("Sound") then
            hookBlueMusic(blue)
        else
            error("Could not find the guiding light music.")
        end

        playerGui.DescendantAdded:Connect(function(obj)
            if obj:IsA("Sound") and obj.Name == "Blue" then
                task.defer(function()
                    local current = findBlueMusic()

                    if current == obj then
                        hookBlueMusic(obj)
                    end
                end)
            end
        end)

        caption("guiding light music replacer loaded!", "info", 5)

    if not ok then
        warn("[Guiding Light Music Replacer] " .. tostring(err))
        caption(
            "guiding light music replacement failed. Check console.",
            "info",
            5
        )
    end

-- =========================================================
-- Despawn Models During Seek Chase
-- =========================================================

local SeekDespawnEnabled = true

local TARGET_PARTS = {
    Rebound = true,
    RushNew = true
}

local seekActive = false
local cleanupInProgress = false

local function getTargetModel(part)
    if not part or not part:IsA("BasePart") then
        return nil
    end

    if not TARGET_PARTS[part.Name] then
        return nil
    end

    local current = part.Parent
    local nearestModel = nil

    while current and current ~= Workspace do
        if current:IsA("Model") then
            nearestModel = current
            break
        end

        current = current.Parent
    end

    if nearestModel then
        local seekClone = Workspace:FindFirstChild(SEEK_NAME)

        if seekClone and (
            nearestModel == seekClone
            or nearestModel:IsDescendantOf(seekClone)
        ) then
            return nil
        end
    end

    return nearestModel
end

local function removeTargetPart(part)
    if not SeekDespawnEnabled then
        return
    end

    if not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    local model = getTargetModel(part)

    if model and model.Parent then
        pcall(function()
            model:Destroy()
        end)
    end
end

local function cleanupExistingTargets()
    if cleanupInProgress or not SeekDespawnEnabled then
        return
    end

    if not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    cleanupInProgress = true

    local success, err = pcall(function()
        local descendants = Workspace:GetDescendants()
        local modelsToRemove = {}

        for _, obj in ipairs(descendants) do
            if obj:IsA("BasePart") and TARGET_PARTS[obj.Name] then
                local model = getTargetModel(obj)

                if model and model.Parent then
                    modelsToRemove[model] = true
                end
            end
        end

        for model in pairs(modelsToRemove) do
            if model.Parent then
                pcall(function()
                    model:Destroy()
                end)
            end
        end
    end)

    cleanupInProgress = false

    if not success then
        warn("[Seek Cleanup] Error:", err)
    end
end

local function updateSeekState()
    local seekClone = Workspace:FindFirstChild(SEEK_NAME)
    local wasActive = seekActive

    seekActive = seekClone ~= nil

    if seekActive and not wasActive then
        task.defer(cleanupExistingTargets)
    end
end

Workspace.ChildAdded:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

Workspace.DescendantAdded:Connect(function(obj)
    if not SeekDespawnEnabled or not seekActive then
        return
    end

    if obj:IsA("BasePart") and TARGET_PARTS[obj.Name] then
        task.defer(function()
            removeTargetPart(obj)
        end)
    end
end)

updateSeekState()

-- =========================================================
-- Door-Based Random Entity Spawning
-- =========================================================

local DepthSpawnChance = 15
local ReboundSpawnChance = 10
local DeerGodSpawnChance = 10

local GameData = ReplicatedStorage:WaitForChild("GameData")
local LatestRoom = GameData:WaitForChild("LatestRoom")

local lastRoom = LatestRoom.Value
local greedDoorCounter = 0

local function rollSpawn(chance, spawnFunction)
    if math.random(1, 100) <= chance then
        task.spawn(spawnFunction)
    end
end

LatestRoom:GetPropertyChangedSignal("Value"):Connect(function()
    local newRoom = LatestRoom.Value

    if newRoom == lastRoom then
        return
    end

    lastRoom = newRoom

    if newRoom >= 5 then
        rollSpawn(DepthSpawnChance, spawnDepth)
    end

    if newRoom >= 12 then
        rollSpawn(ReboundSpawnChance, spawnRebound)
    end

    if newRoom >= 23 then
        rollSpawn(DeerGodSpawnChance, spawnDeerGod)
    end

    if newRoom >= 35 then
        rollSpawn(A60SpawnChance, spawnA60)
    end

    greedDoorCounter += 1

    if greedDoorCounter >= GreedEveryDoors then
        greedDoorCounter = 0
        rollSpawn(GreedSpawnChance, spawnGreed)
    end
end)

-- =========================================================
-- Admin Commands
-- =========================================================

local lastCommand = ""
local lastCommandTime = 0

local function handleCommand(message)
    if not IsAdmin then
        return
    end

    message = tostring(message):lower()
    message = message:match("^%s*(.-)%s*$")

    local now = os.clock()

    if message == lastCommand
        and now - lastCommandTime < 0.25 then
        return
    end

    lastCommand = message
    lastCommandTime = now

    if message == "/depth" then
        spawnDepth()
    elseif message == "/rebound" then
        spawnRebound()
    elseif message == "/deergod" then
        spawnDeerGod()
    elseif message == "/a60" then
        spawnA60()
    elseif message == "/greed" then
        spawnGreed()
    end
end

-- =========================================================
-- TextChatService
-- =========================================================

pcall(function()
    TextChatService.MessageReceived:Connect(function(message)
        if not message.TextSource then
            return
        end

        if message.TextSource.UserId ~= Plr.UserId then
            return
        end

        handleCommand(message.Text)
    end)
end)
