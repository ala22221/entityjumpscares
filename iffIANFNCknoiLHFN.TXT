local SEEK_NAME = "SeekMovingNewClone"
local currentSeek = nil
local seekConnections = {}
local seekPartConnections = {}
local seekSetupToken = 0

local function disconnectSeekConnections()
    for _, connection in ipairs(seekConnections) do
        connection:Disconnect()
    end

    for _, connection in ipairs(seekPartConnections) do
        connection:Disconnect()
    end

    table.clear(seekConnections)
    table.clear(seekPartConnections)
end

local function applySeekChanges(seekClone)
    if not seekClone or not seekClone.Parent then
        return
    end

    local seekRig = seekClone:FindFirstChild("SeekRig")
    local figure = seekClone:FindFirstChild("Figure")

    if figure and figure:IsA("BasePart") then
        figure.Transparency = 0

        local scream = figure:FindFirstChild("Scream")
        local flange = scream and scream:FindFirstChild(
            "FlangeSoundEffect"
        )

        if flange then
            flange.Enabled = false
        end

        local attachment = figure:FindFirstChild("Attachment")
        local emitter = attachment and attachment:FindFirstChild(
            "ParticleEmitter"
        )

        if emitter then
            emitter.Enabled = true
        end

        local camAttach = figure:FindFirstChild("CamAttach")
        local spotlight = camAttach and camAttach:FindFirstChild(
            "SpotLight"
        )

        if spotlight then
            spotlight.Enabled = true
        end
    end

    if seekRig then
        for _, object in ipairs(seekRig:GetDescendants()) do
            if object:IsA("BasePart") then
                object.Transparency = 1
            elseif object:IsA("Decal")
                or object:IsA("Texture") then
                object.Transparency = 1
            end
        end
    end
end

local function watchSeekPart(object, seekClone)
    if not object:IsA("BasePart") then
        return
    end

    local connection = object:GetPropertyChangedSignal(
        "Transparency"
    ):Connect(function()
        if currentSeek ~= seekClone or not seekClone.Parent then
            return
        end

        task.defer(function()
            if currentSeek == seekClone and seekClone.Parent then
                applySeekChanges(seekClone)
            end
        end)
    end)

    table.insert(seekPartConnections, connection)
end

local function setupSeek(seekClone)
    if not seekClone or not seekClone.Parent then
        return
    end

    if currentSeek == seekClone then
        applySeekChanges(seekClone)
        return
    end

    disconnectSeekConnections()

    currentSeek = seekClone
    seekSetupToken += 1

    local setupToken = seekSetupToken

    applySeekChanges(seekClone)

    for _, object in ipairs(seekClone:GetDescendants()) do
        watchSeekPart(object, seekClone)
    end

    table.insert(
        seekConnections,
        seekClone.DescendantAdded:Connect(function(object)
            if currentSeek ~= seekClone then
                return
            end

            watchSeekPart(object, seekClone)

            task.defer(function()
                if currentSeek == seekClone and seekClone.Parent then
                    applySeekChanges(seekClone)
                end
            end)
        end)
    )

    table.insert(
        seekConnections,
        seekClone.AncestryChanged:Connect(function()
            if currentSeek == seekClone and not seekClone.Parent then
                currentSeek = nil
                disconnectSeekConnections()
            end
        end)
    )

    task.spawn(function()
        for _ = 1, 10 do
            if currentSeek ~= seekClone
                or not seekClone.Parent
                or setupToken ~= seekSetupToken then
                return
            end

            applySeekChanges(seekClone)
            task.wait(0.1)
        end
    end)
end

Workspace.ChildAdded:Connect(function(child)
    if child.Name == SEEK_NAME then
        setupSeek(child)
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child == currentSeek then
        currentSeek = nil
        seekSetupToken += 1
        disconnectSeekConnections()
    end
end)

local existingSeek = Workspace:FindFirstChild(SEEK_NAME)

if existingSeek then
    setupSeek(existingSeek)
end

local AchievementGiver = nil
local givenAchievements = {}
local achievementsInProgress = {}

local AchievementURL =
    "https://raw.githubusercontent.com/ala22221/Backups-of-some-stuff/refs/heads/main/Fixed%20Achievements%20by%20DripCapybara.lua"

local function loadAchievementGiver()
    if type(AchievementGiver) == "function" then
        return true
    end

    local ok, result = pcall(function()
        local source = game:HttpGet(AchievementURL)
        local loader, compileError = loadstring(source)

        if not loader then
            error(compileError or "Compilation failed")
        end

        return loader()
    end)

    if not ok then
        warn("[Achievements] Loader error:", result)
        return false
    end

    if type(result) ~= "function" then
        warn(
            "[Achievements] Expected a function, got:",
            typeof(result)
        )
        return false
    end

    AchievementGiver = result
    print("[Achievements] Giver loaded successfully.")

    return true
end

local function waitForAchievementUI(timeout)
    local deadline = os.clock() + (timeout or 60)

    while os.clock() < deadline do
        local playerGui = Plr:FindFirstChildOfClass("PlayerGui")
        local globalUI = playerGui
            and playerGui:FindFirstChild("GlobalUI")
        local holder = globalUI
            and globalUI:FindFirstChild("AchievementsHolder")
        local achievement = holder
            and holder:FindFirstChild("Achievement")

        if achievement then
            return true
        end

        task.wait(0.5)
    end

    return false
end

local function giveAchievement(id, data)
    if givenAchievements[id] or achievementsInProgress[id] then
        return
    end

    achievementsInProgress[id] = true

    task.spawn(function()
        local delivered = false

        for attempt = 1, 10 do
            if givenAchievements[id] then
                delivered = true
                break
            end

            if not Plr.Parent then
                break
            end

            if not loadAchievementGiver() then
                warn("[Achievements] Retry", attempt, "for", id)
                task.wait(2)
                continue
            end

            if not waitForAchievementUI(10) then
                warn(
                    "[Achievements] UI not ready for",
                    id,
                    "attempt",
                    attempt
                )

                task.wait(1)
                continue
            end

            local ok, err = pcall(function()
                AchievementGiver(data)
            end)

            if ok then
                givenAchievements[id] = true
                delivered = true

                print("[Achievements] Giver called for:", id)
                break
            else
                warn(
                    "[Achievements] Failed to give",
                    id,
                    tostring(err)
                )

                task.wait(1)
            end
        end

        achievementsInProgress[id] = nil

        if not delivered then
            warn(
                "[Achievements] Failed after retries:",
                id,
                "- it can be requested again."
            )
        end
    end)
end

task.spawn(function()
    for _ = 1, 10 do
        if loadAchievementGiver() then
            return
        end

        task.wait(2)
    end

    warn("[Achievements] Could not load the achievement giver.")
end)

local SeekMusicURL =
    "https://raw.githubusercontent.com/ala22221/sounds/main/106.ogg"

local SeekMusicFolder = "CustomMusic"
local SeekMusicFile = SeekMusicFolder .. "/106.ogg"
local SeekMusicVolume = 1

task.spawn(function()
    if not (
        writefile and isfile and getcustomasset
        and makefolder and isfolder
    ) then
        caption("Executor doesn't support custom audio.", "info", 5)
        return
    end

    local ok, err = pcall(function()
        if not isfolder(SeekMusicFolder) then
            makefolder(SeekMusicFolder)
        end

        if not isfile(SeekMusicFile) then
            local success, data = pcall(function()
                return game:HttpGet(SeekMusicURL)
            end)

            if not success or not data or #data == 0 then
                caption("Seek music download failed.", "info", 5)
                return
            end

            writefile(SeekMusicFile, data)
        end

        local assetId = getcustomasset(SeekMusicFile)

        local floorRep = ReplicatedStorage:WaitForChild(
            "FloorReplicated",
            30
        )

        if not floorRep then
            return
        end

        local hookedSounds = {}

        local function hookSound(sound)
            if not sound:IsA("Sound") or hookedSounds[sound] then
                return
            end

            hookedSounds[sound] = true

            local function applyReplacement()
                if not sound.Parent then
                    return
                end

                sound.SoundId = assetId
                sound.Volume = SeekMusicVolume
            end

            applyReplacement()

            sound:GetPropertyChangedSignal("SoundId"):Connect(
                applyReplacement
            )

            sound:GetPropertyChangedSignal("Volume"):Connect(
                applyReplacement
            )
        end

        local seekMusic = floorRep:FindFirstChild("SeekMusic")

        if seekMusic then
            hookSound(seekMusic)
        end

        floorRep.ChildAdded:Connect(function(child)
            if child.Name == "SeekMusic" then
                task.defer(function()
                    hookSound(child)
                end)
            end
        end)

        local function checkClone(obj)
            if obj:IsA("Sound") and obj.Name == "SeekMusic" then
                task.defer(function()
                    hookSound(obj)
                end)
            end
        end

        Workspace.DescendantAdded:Connect(checkClone)
        SoundService.DescendantAdded:Connect(checkClone)

        caption("music replacer loaded!", "info", 5)
    end)

    if not ok then
        warn("[Seek Music Replacer]", err)
    end
end)

-- =========================================================
-- Seek Chase Cleanup
-- =========================================================

local SeekDespawnEnabled = true

local TARGET_PARTS = {
    Rebound = true,
    RushNew = true
}

local seekActive = false
local cleanupInProgress = false

local function getTargetModel(part)
    if not part or not part:IsA("BasePart")
        or not TARGET_PARTS[part.Name] then
        return nil
    end

    local current = part.Parent

    while current and current ~= Workspace do
        if current:IsA("Model") then
            local seekClone = Workspace:FindFirstChild(SEEK_NAME)

            if seekClone and (
                current == seekClone
                or current:IsDescendantOf(seekClone)
            ) then
                return nil
            end

            return current
        end

        current = current.Parent
    end

    return nil
end

local function removeTargetPart(part)
    if not SeekDespawnEnabled
        or not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    local model = getTargetModel(part)

    if model and model.Parent then
        pcall(function()
            model:Destroy()
        end)
    end
end

local function cleanupExistingTargets()
    if cleanupInProgress or not SeekDespawnEnabled
        or not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    cleanupInProgress = true

    local ok, err = pcall(function()
        local modelsToRemove = {}

        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") and TARGET_PARTS[obj.Name] then
                local model = getTargetModel(obj)

                if model then
                    modelsToRemove[model] = true
                end
            end
        end

        for model in pairs(modelsToRemove) do
            if model.Parent then
                model:Destroy()
            end
        end
    end)

    cleanupInProgress = false

    if not ok then
        warn("[Seek Cleanup]", err)
    end
end

local function updateSeekState()
    local wasActive = seekActive
    seekActive = Workspace:FindFirstChild(SEEK_NAME) ~= nil

    if seekActive and not wasActive then
        task.defer(cleanupExistingTargets)
    end
end

Workspace.ChildAdded:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

Workspace.DescendantAdded:Connect(function(obj)
    if SeekDespawnEnabled and seekActive
        and obj:IsA("BasePart")
        and TARGET_PARTS[obj.Name] then
        task.defer(function()
            removeTargetPart(obj)
        end)
    end
end)

updateSeekState()

local function createEntity(config, label)
    local ok, entityTable = pcall(function()
        return Spawner.createEntity(config)
    end)

    if not ok or not entityTable then
        warn("[" .. label .. "] Failed to create entity:", entityTable)
        return nil

    return entityTable
end

local function runEntity(entityTable, label)
    task.spawn(function()
        local ok, err = pcall(function()
            Spawner.runEntity(entityTable)
        end)

        if not ok then
            warn("[" .. label .. "] Failed to run entity:", err)
        end
    end)
end

local function setDebugCallbacks(entityTable, callbacks)
    if not entityTable.Debug then
        warn("[Achievements] Debug callbacks unavailable for entity.")
        return
    end

    for name, callback in pairs(callbacks) do
        entityTable.Debug[name] = callback
    end
end

-- =========================================================
-- Door-Based Random Spawning
-- =========================================================

local GameData = ReplicatedStorage:WaitForChild("GameData")
local LatestRoom = GameData:WaitForChild("LatestRoom")

local lastRoom = LatestRoom.Value
local greedDoorCounter = 0

local function rollSpawn(chance, spawnFunction)
    if math.random(1, 100) <= chance then
        task.spawn(spawnFunction)
    end
end

LatestRoom:GetPropertyChangedSignal("Value"):Connect(function()
    local newRoom = LatestRoom.Value

    if newRoom == lastRoom then
        return
    end

    lastRoom = newRoom

    if newRoom >= 5 then
        rollSpawn(DepthSpawnChance, spawnDepth)
    end

    if newRoom >= 12 then
        rollSpawn(ReboundSpawnChance, spawnRebound)
    end

    if newRoom >= 23 then
        rollSpawn(DeerGodSpawnChance, spawnDeerGod)
    end

    if newRoom >= 35 then
        rollSpawn(A60SpawnChance, spawnA60)
    end

    greedDoorCounter += 1

    if greedDoorCounter >= GreedEveryDoors then
        greedDoorCounter = 0
        rollSpawn(GreedSpawnChance, spawnGreed)
    end
end)

-- =========================================================
-- Admin Commands
-- =========================================================

local lastCommand = ""
local lastCommandTime = 0

local function handleCommand(message)
    if not IsAdmin then
        return
    end

    message = tostring(message):lower()
    message = message:match("^%s*(.-)%s*$")

    local now = os.clock()

    if message == lastCommand and now - lastCommandTime < 0.25 then
        return
    end

    lastCommand = message
    lastCommandTime = now

    if message == "/depth" then
        spawnDepth()
    elseif message == "/rebound" then
        spawnRebound()
    elseif message == "/deergod" then
        spawnDeerGod()
    elseif message == "/a60" then
        spawnA60()
    elseif message == "/greed" then
        spawnGreed()
    end
end

-- =========================================================
-- TextChatService
-- =========================================================

pcall(function()
    TextChatService.MessageReceived:Connect(function(message)
        if not message.TextSource then
            return
        end

        if message.TextSource.UserId ~= Plr.UserId then
            return
        end

        handleCommand(message.Text)
    end)
end)

-- =========================================================
-- Legacy Chat
-- =========================================================

pcall(function()
    Plr.Chatted:Connect(function(message)
        handleCommand(message)
    end)
end)
