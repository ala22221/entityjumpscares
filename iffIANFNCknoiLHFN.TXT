-- =========================================================

local HIDE_SCARE_SOUND_ID = "rbxassetid://5263560896"
local SCREECH_CAUGHT_SOUND_ID = "rbxassetid://297569488"
local SCREECH_ATTACK_SOUND_ID = "rbxassetid://18967004856"
local SCREECH_EYE_COLOR = Color3.fromRGB(255, 0, 0)

local function findDescendantByName(root, wantedName)
    if not root then
        return nil
    end

    local direct = root:FindFirstChild(wantedName)
    if direct then
        return direct
    end

    return root:FindFirstChild(wantedName, true)
end

local function setSoundId(root, soundName, soundId)
    local sound = findDescendantByName(root, soundName)
    if sound and sound:IsA("Sound") then
        sound.SoundId = soundId
    end
end

local function applyHideAndScreechChanges()
    local playerGui = Plr:FindFirstChildOfClass("PlayerGui")
    local mainUI = playerGui and playerGui:FindFirstChild("MainUI")
    local initiator = mainUI and mainUI:FindFirstChild("Initiator")
    local mainGame = initiator and initiator:FindFirstChild("Main_Game")
    local remoteListener = mainGame and mainGame:FindFirstChild("RemoteListener")
    local modules = remoteListener and remoteListener:FindFirstChild("Modules")

    -- Hide Monster scare sound
    local hideModule = modules and findDescendantByName(modules, "HideMonster")
    local hideScare = hideModule and findDescendantByName(hideModule, "Scare")
    if hideScare and hideScare:IsA("Sound") then
        hideScare.SoundId = HIDE_SCARE_SOUND_ID
    end

    -- Screech module sounds (module layout differs between DOORS versions,
    -- so search recursively inside the Modules folder).
    local screechModule = modules and findDescendantByName(modules, "Screech")
    if screechModule then
        setSoundId(screechModule, "Caught", SCREECH_CAUGHT_SOUND_ID)
        setSoundId(screechModule, "Attack", SCREECH_ATTACK_SOUND_ID)
    end

    -- Make Screech eye parts/decals red when a Screech instance is present.
    local function colorScreechEyes(root)
        if not root then return end
        for _, obj in ipairs(root:GetDescendants()) do
            local lowerName = string.lower(obj.Name)
            if (lowerName == "eye" or lowerName == "eyes" or lowerName == "screech eyes") then
                if obj:IsA("BasePart") then
                    obj.Color = SCREECH_EYE_COLOR
                elseif obj:IsA("Decal") or obj:IsA("Texture") then
                    obj.Color3 = SCREECH_EYE_COLOR
                end
            end
        end
    end

    colorScreechEyes(Workspace)
end

task.spawn(function()
    while task.wait(1) do
        pcall(applyHideAndScreechChanges)
    end
end)

Workspace.DescendantAdded:Connect(function(obj)
    local lowerName = string.lower(obj.Name)
    if lowerName == "screech" or lowerName == "eye" or lowerName == "eyes" then
        task.defer(function()
            pcall(applyHideAndScreechChanges)
        end)
    end
end)

Plr:WaitForChild("PlayerGui").DescendantAdded:Connect(function(obj)
    local lowerName = string.lower(obj.Name)
    if lowerName == "hideMonster" or lowerName == "screech" or lowerName == "scare" or lowerName == "caught" or lowerName == "attack" then
        task.defer(function()
            pcall(applyHideAndScreechChanges)
        end)
    end
end)

-- =========================================================
-- Achievement System (retry-safe)
-- =========================================================

local AchievementGiver
-- Only blocks simultaneous duplicate requests. Achievements may display again later.
local achievementsInProgress = {}

local AchievementURL =
    "https://raw.githubusercontent.com/ala22221/Backups-of-some-stuff/refs/heads/main/Fixed%20Achievements%20by%20DripCapybara.lua"

local function loadAchievementGiver()
    if type(AchievementGiver) == "function" then
        return true
    end

    for attempt = 1, 5 do
        local ok, result = pcall(function()
            local source = game:HttpGet(AchievementURL)
            local loader, compileError = loadstring(source)
            if not loader then
                error(compileError or "Achievement module failed to compile")
            end
            return loader()
        end)

        if ok and type(result) == "function" then
            AchievementGiver = result
            return true
        end

        warn("[Easy'nt Mode] Achievement module load attempt " .. attempt .. " failed: " .. tostring(result))
        task.wait(1)
    end

    return false
end

local function waitForAchievementUI(timeout)
    timeout = timeout or 10
    local deadline = os.clock() + timeout

    while os.clock() < deadline do
        local playerGui = Plr:FindFirstChildOfClass("PlayerGui")
        local globalUI = playerGui and playerGui:FindFirstChild("GlobalUI")
        local holder = globalUI and globalUI:FindFirstChild("AchievementsHolder")
        local template = holder and holder:FindFirstChild("Achievement")

        if template then
            return true
        end

        task.wait(0.25)
    end

    return false
end

local function giveAchievement(id, data)
    if achievementsInProgress[id] then
        return
    end

    achievementsInProgress[id] = true

    task.spawn(function()
        local succeeded = false

        for attempt = 1, 30 do
            if not loadAchievementGiver() then
                task.wait(1)
            elseif waitForAchievementUI(2) then
                local ok, err = pcall(function()
                    AchievementGiver(data)
                end)

                if ok then
                    succeeded = true
                    break
                end

                warn("[Easy'nt Mode] Achievement '" .. tostring(id) .. "' attempt " .. attempt .. " failed: " .. tostring(err))
                task.wait(0.5)
            else
                task.wait(0.5)
            end
        end

        if not succeeded then
            warn("[Easy'nt Mode] Could not display achievement: " .. tostring(id))
        end

        achievementsInProgress[id] = nil
    end)
end

local SeekMusicURL =
    "https://raw.githubusercontent.com/ala22221/sounds/main/106.ogg"

local SeekMusicFolder = "CustomMusic"
local SeekMusicFile = SeekMusicFolder .. "/106.ogg"
local SeekMusicVolume = 1

task.spawn(function()

    if not (writefile and isfile and getcustomasset
        and makefolder and isfolder) then

        caption(
            "Executor doesn't support custom audio.",
            "info",
            5
        )

        return
    end

    pcall(function()

        if not isfolder(SeekMusicFolder) then
            makefolder(SeekMusicFolder)
        end

        -- Download once (delete the file from workspace to re-download)

        if not isfile(SeekMusicFile) then

            local ok, data = pcall(function()
                return game:HttpGet(SeekMusicURL)
            end)

            if not ok or not data or #data == 0 then

                caption(
                    "Seek music download failed.",
                    "info",
                    5
                )

                return
            end

            writefile(SeekMusicFile, data)

        end

        local assetId = getcustomasset(SeekMusicFile)

        local floorRep =
            ReplicatedStorage:WaitForChild(
                "FloorReplicated",
                30
            )

        if not floorRep then
            return
        end

        local function hookSound(sound)

            if not sound
                or not sound:IsA("Sound") then
                return
            end

            sound.SoundId = assetId
            sound.Volume = SeekMusicVolume

            -- Re-apply if the game resets them

            sound:GetPropertyChangedSignal("SoundId"):Connect(
                function()

                    if sound.SoundId ~= assetId then
                        sound.SoundId = assetId
                    end

                end
            )

            sound:GetPropertyChangedSignal("Volume"):Connect(
                function()

                    if sound.Volume ~= SeekMusicVolume then
                        sound.Volume = SeekMusicVolume
                    end

                end
            )

        end

        local seekMusic =
            floorRep:FindFirstChild("SeekMusic")

        if seekMusic then
            hookSound(seekMusic)
        end

        -- If the game re-creates it

        floorRep.ChildAdded:Connect(function(child)

            if child.Name == "SeekMusic" then
                task.wait()
                hookSound(child)
            end

        end)

        -- If the game clones it elsewhere (Workspace / SoundService)

        local function checkClone(obj)

            if obj:IsA("Sound")
                and obj.Name == "SeekMusic" then

                task.wait()
                hookSound(obj)

            end

        end

        Workspace.DescendantAdded:Connect(checkClone)
        SoundService.DescendantAdded:Connect(checkClone)

        caption(
            "music replacer loaded!",
            "info",
            5
        )

    end)

end)


-- =========================================================
-- Blue Health Music Replacer (Intro.ogg)
-- Downloads once into the executor's file workspace and
-- replaces PlayerGui.MainUI.Initiator.Main_Game.Health.Music.Blue.
-- =========================================================

local BlueMusicURL = "https://raw.githubusercontent.com/ala22221/sounds/main/Intro.ogg"
local BlueMusicFolder = "CustomMusic"
local BlueMusicFile = BlueMusicFolder .. "/Intro.ogg"
local BlueMusicVolume = 1
local BlueMusicLooped = true

task.spawn(function()
    if not (writefile and isfile and getcustomasset and makefolder and isfolder) then
        caption("Executor doesn't support custom audio.", "info", 5)
        return
    end

    local ok, err = pcall(function()
        if not isfolder(BlueMusicFolder) then
            makefolder(BlueMusicFolder)
        end

        -- Download only when the cached file is missing.
        if not isfile(BlueMusicFile) then
            local downloadOK, data = pcall(function()
                return game:HttpGet(BlueMusicURL)
            end)

            if not downloadOK or not data or #data == 0 then
                error("Intro.ogg download failed.")
            end

            writefile(BlueMusicFile, data)
        end

        local assetId = getcustomasset(BlueMusicFile)

        local function findBlueMusic()
            local playerGui = Plr:FindFirstChildOfClass("PlayerGui")
            if not playerGui then
                return nil
            end

            local mainUI = playerGui:FindFirstChild("MainUI")
            local initiator = mainUI and mainUI:FindFirstChild("Initiator")
            local mainGame = initiator and initiator:FindFirstChild("Main_Game")
            local health = mainGame and mainGame:FindFirstChild("Health")
            local music = health and health:FindFirstChild("Music")
            local blue = music and music:FindFirstChild("Blue")

            if blue and blue:IsA("Sound") then
                return blue
            end

            return nil
        end

        local hookedSounds = {}

        local function hookBlueMusic(sound)
            if not sound or not sound:IsA("Sound") or hookedSounds[sound] then
                return
            end

            hookedSounds[sound] = true

            local function applyReplacement()
                if not sound.Parent then
                    return
                end

                if sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end

                if sound.Volume ~= BlueMusicVolume then
                    sound.Volume = BlueMusicVolume
                end

                if sound.Looped ~= BlueMusicLooped then
                    sound.Looped = BlueMusicLooped
                end
            end

            applyReplacement()

            sound:GetPropertyChangedSignal("SoundId"):Connect(function()
                if sound.Parent and sound.SoundId ~= assetId then
                    sound.SoundId = assetId
                end
            end)

            sound:GetPropertyChangedSignal("Volume"):Connect(function()
                if sound.Parent and sound.Volume ~= BlueMusicVolume then
                    sound.Volume = BlueMusicVolume
                end
            end)

            sound:GetPropertyChangedSignal("Looped"):Connect(function()
                if sound.Parent and sound.Looped ~= BlueMusicLooped then
                    sound.Looped = BlueMusicLooped
                end
            end)

            -- Start playing after applying the local replacement.
            pcall(function()
                sound:Play()
            end)
        end

        -- Wait for the existing Blue sound, if it hasn't loaded yet.
        local playerGui = Plr:WaitForChild("PlayerGui", 30)
        if not playerGui then
            error("PlayerGui did not load.")
        end

        local blue = findBlueMusic()
        if not blue then
            local mainUI = playerGui:WaitForChild("MainUI", 30)
            local initiator = mainUI and mainUI:WaitForChild("Initiator", 30)
            local mainGame = initiator and initiator:WaitForChild("Main_Game", 30)
            local health = mainGame and mainGame:WaitForChild("Health", 30)
            local music = health and health:WaitForChild("Music", 30)
            blue = music and music:WaitForChild("Blue", 30)
        end

        if blue and blue:IsA("Sound") then
            hookBlueMusic(blue)
        else
            error("Could not find the guiding light music.")
        end

        -- Re-apply if the game replaces the Blue sound instance.
        playerGui.DescendantAdded:Connect(function(obj)
            if obj:IsA("Sound") and obj.Name == "Blue" then
                task.defer(function()
                    local current = findBlueMusic()
                    if current == obj then
                        hookBlueMusic(obj)
                    end
                end)
            end
        end)

        caption("guiding light music replacer loaded!", "info", 5)
    end)

    if not ok then
        warn("[Guiding Light Music Replacer] " .. tostring(err))
        caption("guiding light music replacement failed. Check console.", "info", 5)
    end
end)

-- =========================================================
-- Despawn Models During Seek Chase (Optimized)
-- =========================================================

local SeekDespawnEnabled = true
local Workspace = game:GetService("Workspace")

local SEEK_NAME = "SeekMovingNewClone"
local TARGET_PARTS = {
    Rebound = true,
    RushNew = true
}

local seekActive = false
local cleanupInProgress = false

-- Find the nearest Model containing a targeted part.
local function getTargetModel(part)
    if not part or not part:IsA("BasePart") then
        return nil
    end

    if not TARGET_PARTS[part.Name] then
        return nil
    end

    local current = part.Parent
    local nearestModel = nil

    while current and current ~= Workspace do
        if current:IsA("Model") then
            nearestModel = current
            break
        end

        current = current.Parent
    end

    -- Never destroy Seek itself or anything inside Seek.
    if nearestModel then
        local seekClone = Workspace:FindFirstChild(SEEK_NAME)

        if seekClone and (
            nearestModel == seekClone
            or nearestModel:IsDescendantOf(seekClone)
        ) then
            return nil
        end
    end

    return nearestModel
end

local function removeTargetPart(part)
    if not SeekDespawnEnabled then
        return
    end

    if not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    local model = getTargetModel(part)

    if model and model.Parent then
        pcall(function()
            model:Destroy()
        end)
    end
end

-- One full scan when Seek appears.
-- This replaces the expensive repeating Workspace scan.
local function cleanupExistingTargets()
    if cleanupInProgress or not SeekDespawnEnabled then
        return
    end

    if not Workspace:FindFirstChild(SEEK_NAME) then
        return
    end

    cleanupInProgress = true

    local success, err = pcall(function()
        local descendants = Workspace:GetDescendants()
        local modelsToRemove = {}

        for _, obj in ipairs(descendants) do
            if obj:IsA("BasePart") and TARGET_PARTS[obj.Name] then
                local model = getTargetModel(obj)

                if model and model.Parent then
                    modelsToRemove[model] = true
                end
            end
        end

        for model in pairs(modelsToRemove) do
            if model.Parent then
                pcall(function()
                    model:Destroy()
                end)
            end
        end
    end)

    cleanupInProgress = false

    if not success then
        warn("[Seek Cleanup] Error:", err)
    end
end

-- Detect Seek appearing or disappearing.
local function updateSeekState()
    local seekClone = Workspace:FindFirstChild(SEEK_NAME)
    local wasActive = seekActive

    seekActive = seekClone ~= nil

    if seekActive and not wasActive then
        task.defer(cleanupExistingTargets)
    end
end

Workspace.ChildAdded:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child.Name == SEEK_NAME then
        updateSeekState()
    end
end)

-- Only react to newly added parts with the exact target names.
-- Do not rescan Workspace for unrelated additions.
Workspace.DescendantAdded:Connect(function(obj)
    if not SeekDespawnEnabled or not seekActive then
        return
    end

    if obj:IsA("BasePart") and TARGET_PARTS[obj.Name] then
        task.defer(function()
            removeTargetPart(obj)
        end)
    end
end)

-- Handle Seek already existing when this section starts.
updateSeekState()

-- =========================================================
-- Door-Based Random Entity Spawning
-- =========================================================

-- Chances are percentages. Change these values if desired.
local DepthSpawnChance = 15
local ReboundSpawnChance = 10
local DeerGodSpawnChance = 10
-- A60SpawnChance and GreedSpawnChance are defined in Spawn Chances above.

-- Use the same full LatestRoom path as the original Greed door handler.
local GameData = ReplicatedStorage:WaitForChild("GameData")
local LatestRoom = GameData:WaitForChild("LatestRoom")

local lastRoom = LatestRoom.Value
local greedDoorCounter = 0

local function rollSpawn(chance, spawnFunction)
    if math.random(1, 100) <= chance then
        task.spawn(spawnFunction)
    end
end

-- Every change to GameData.LatestRoom.Value represents a newly reached room.
-- All automatic entity spawning is handled here; no entity spawn timers remain.
LatestRoom:GetPropertyChangedSignal("Value"):Connect(function()
    local newRoom = LatestRoom.Value

    if newRoom == lastRoom then
        return
    end

    lastRoom = newRoom

    -- Do not even roll an entity's chance until its minimum door is reached.
    if newRoom >= 5 then
        rollSpawn(DepthSpawnChance, spawnDepth)
    end

    if newRoom >= 12 then
        rollSpawn(ReboundSpawnChance, spawnRebound)
    end

    if newRoom >= 23 then
        rollSpawn(DeerGodSpawnChance, spawnDeerGod)
    end

    if newRoom >= 35 then
        rollSpawn(A60SpawnChance, spawnA60)
    end

    -- Preserve Greed's original behavior: one 20% roll every 5 door changes.
    greedDoorCounter = greedDoorCounter + 1
    if greedDoorCounter >= GreedEveryDoors then
        greedDoorCounter = 0
        rollSpawn(GreedSpawnChance, spawnGreed)
    end
end)

-- =========================================================
-- Admin Commands
-- =========================================================

local lastCommand = ""
local lastCommandTime = 0

local function handleCommand(message)

    if not IsAdmin then
        return
    end

    message =
        tostring(message):lower()

    message =
        message:match("^%s*(.-)%s*$")

    local now = os.clock()

    if message == lastCommand
        and now - lastCommandTime < 0.25 then

        return
    end

    lastCommand = message
    lastCommandTime = now

    if message == "/depth" then

        spawnDepth()

    elseif message == "/rebound" then

        spawnRebound()

    elseif message == "/deergod" then

        spawnDeerGod()

    elseif message == "/a60" then

        spawnA60()

    elseif message == "/greed" then

        spawnGreed()

    end

end

-- TextChatService
-- =========================================================

pcall(function()

    TextChatService.MessageReceived:Connect(
        function(message)

            if not message.TextSource then
                return
            end

            if message.TextSource.UserId
                ~= Plr.UserId then

                return
            end

            handleCommand(
                message.Text
            )

        end
    )

end)
